очистить логи:

sudo journalctl --vacuum-time=1s
sudo find /var/log -type f -name "*.log" -exec truncate -s 0 {} \;
sudo find /var/log -type f \( -name "*.gz" -o -name "*.old" -o -name "*.1" \) -delete

очистить ситорию:

history -c
history -w
> /root/.bash_history
